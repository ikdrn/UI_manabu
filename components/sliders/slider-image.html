スライダー・カルーセル

画像カルーセル

横スクロールで画像を切り替えるカルーセル。CSSのscroll-snapを活用することで、タッチ操作にも対応したスムーズなスライドを実現します。

設計の意図

なぜscroll-snapを使うのか？
従来のカルーセルはtransform: translateX()で位置を手動計算していましたが、scroll-snapを使えばブラウザがネイティブにスナップ位置を管理します。慣性スクロールもブラウザ任せになるため、モバイルでの操作感が格段に向上し、アクセシビリティも標準のスクロール動作に準拠します。

IntersectionObserver vs scrollイベント
scrollイベントは毎フレーム発火するため、パフォーマンスに悪影響を及ぼす可能性があります。IntersectionObserverは要素の可視状態が変わった時だけコールバックが実行されるため、ドットの更新を効率的に行えます。

タッチ・スワイプ対応が無料で付いてくる
scroll-snapはブラウザのネイティブスクロールを利用するため、タッチデバイスでのスワイプ操作が追加のJSなしで自然に動作します。Hammer.jsなどのタッチライブラリは不要です。

矢印キーによるアクセシビリティ
矢印ボタンにはaria-labelを設定し、スクリーンリーダーで「前のスライド」「次のスライド」と読み上げられます。また、カルーセル全体にrole="region"とaria-roledescription="カルーセル"を付与して、支援技術がコンポーネントの役割を正しく認識できるようにしています。

コード解説

/* --- カルーセル全体のコンテナ --- */
.carousel {
  /* position: relative → 矢印ボタンの絶対配置の基準にする */
  position: relative;

  /* width: 100% → 親要素の幅いっぱいに広がる */
  width: 100%;

  /* max-width: 640px → 大きい画面でも広がりすぎないよう制限 */
  max-width: 640px;

  /* margin: 0 auto → 左右均等の余白で中央寄せ */
  margin: 0 auto;

  /* border-radius → 角を丸くして柔らかい印象にする */
  border-radius: var(--radius-md);

  /* overflow: hidden → はみ出したスライドを隠す */
  overflow: hidden;
}

/* --- スクロールトラック（スライドが横並びする領域） --- */
.carousel__track {
  /* display: flex → 子要素（スライド）を横一列に並べる */
  display: flex;

  /* overflow-x: auto → 横方向にスクロール可能にする */
  overflow-x: auto;

  /* scroll-snap-type: x mandatory → 横方向のスナップを強制する。
     mandatory はスクロールが必ずスナップ位置に止まることを保証する */
  scroll-snap-type: x mandatory;

  /* scroll-behavior: smooth → JSからのscrollTo()をなめらかにアニメーション */
  scroll-behavior: smooth;

  /* -webkit-overflow-scrolling: touch → iOS Safariで慣性スクロールを有効化 */
  -webkit-overflow-scrolling: touch;

  /* scrollbar-width: none → Firefox用：スクロールバーを非表示にする */
  scrollbar-width: none;
}

/* WebKit系ブラウザでスクロールバーを非表示にする */
.carousel__track::-webkit-scrollbar {
  display: none;
}

/* --- 各スライド --- */
.carousel__slide {
  /* flex: 0 0 100% → 縮まず広がらず、親の幅ぴったりに固定。
     0 0 100% は grow shrink basis の順 */
  flex: 0 0 100%;

  /* scroll-snap-align: start → このスライドの左端がスナップ位置になる */
  scroll-snap-align: start;

  /* aspect-ratio: 16/9 → 画像の一般的な横長比率を維持 */
  aspect-ratio: 16 / 9;

  /* 中央寄せのフレックスボックスでテキストを配置 */
  display: flex;
  align-items: center;
  justify-content: center;

  /* スライド上のテキストスタイル */
  color: white;
  font-size: var(--text-lg);
  font-weight: var(--font-semibold);

  /* user-select: none → テキスト選択を防止（スワイプ時の誤選択防止） */
  user-select: none;
}

/* --- 矢印ボタン --- */
.carousel__arrow {
  /* position: absolute → カルーセル内で自由配置 */
  position: absolute;

  /* top: 50% + transform → 縦方向の中央に配置 */
  top: 50%;
  transform: translateY(-50%);

  /* 円形ボタンのサイズ */
  width: 40px;
  height: 40px;

  /* border: none → ブラウザデフォルトの枠を消す */
  border: none;

  /* border-radius: 50% → 完全な円形にする */
  border-radius: 50%;

  /* 半透明の黒背景で視認性を確保 */
  background: rgba(0, 0, 0, 0.45);
  color: white;
  font-size: 18px;

  /* cursor: pointer → クリック可能であることを示す */
  cursor: pointer;

  /* フレックスボックスで矢印記号を中央配置 */
  display: flex;
  align-items: center;
  justify-content: center;

  /* なめらかなホバー遷移 */
  transition: background var(--duration-fast);

  /* z-index: 2 → スライドの上に表示する */
  z-index: 2;
}

/* 左矢印の位置 */
.carousel__arrow--prev { left: 12px; }

/* 右矢印の位置 */
.carousel__arrow--next { right: 12px; }

/* --- ドットナビゲーション --- */
.carousel__dots {
  /* 横並びで中央寄せ */
  display: flex;
  justify-content: center;
  gap: 8px;
  padding: 12px 0;
}

.carousel__dot {
  /* 丸いドットのサイズ */
  width: 10px;
  height: 10px;
  border-radius: 50%;
  border: none;
  padding: 0;

  /* 非アクティブ時はグレー */
  background: var(--color-border);
  cursor: pointer;

  /* 色と大きさの変化をなめらかに */
  transition: background var(--duration-fast),
              transform var(--duration-fast);
}

/* is-active → 現在表示中のスライドに対応するドット */
.carousel__dot.is-active {
  /* ブランドカラーで強調 */
  background: var(--color-brand-500);

  /* 1.25倍に拡大して目立たせる */
  transform: scale(1.25);
}
// --- 要素の取得 ---
// querySelector → CSSセレクタで要素を1つ取得する
const track = document.querySelector('#carouselTrack');
// querySelectorAll → 一致する要素をすべて取得（NodeList）
const slides = track.querySelectorAll('.carousel__slide');
const dots = document.querySelectorAll('.carousel__dot');
const prevBtn = document.querySelector('#prevBtn');
const nextBtn = document.querySelector('#nextBtn');

// --- 現在のスライド番号を追跡する変数 ---
let currentIndex = 0;

// --- ドットの表示を更新する関数 ---
function updateDots(index) {
  // forEach → 配列やNodeListの各要素に対して関数を実行する
  dots.forEach((dot, i) => {
    // classList.toggle → 条件がtrueならクラスを追加、falseなら削除
    dot.classList.toggle('is-active', i === index);
    // aria-selected → スクリーンリーダーに選択状態を伝える属性
    dot.setAttribute('aria-selected', i === index);
  });
  // 現在のインデックスを保存
  currentIndex = index;
}

// --- 指定インデックスのスライドまでスクロールする関数 ---
function scrollToSlide(index) {
  // scrollTo → 要素のスクロール位置を指定座標に移動する
  // behavior: 'smooth' → なめらかにアニメーションする
  track.scrollTo({
    left: slides[index].offsetLeft, // offsetLeft → 要素の左端位置
    behavior: 'smooth'
  });
}

// --- IntersectionObserverでスライドの可視状態を監視 ---
// IntersectionObserver → 要素がビューポートに入った/出たを検知するAPI
const observer = new IntersectionObserver(
  (entries) => {
    // entries → 監視対象要素の状態変化の配列
    entries.forEach((entry) => {
      if (entry.isIntersecting) {
        // isIntersecting → 要素が表示領域に入っているかどうか
        // Array.from → NodeListを配列に変換する
        const index = Array.from(slides).indexOf(entry.target);
        updateDots(index);
      }
    });
  },
  {
    root: track,        // root → 交差判定の基準となる要素（スクロール領域）
    threshold: 0.5      // threshold → 50%以上見えたら「交差した」と判定
  }
);

// 各スライドを監視対象に登録する
slides.forEach((slide) => observer.observe(slide));

// --- 矢印ボタンのクリックイベント ---
prevBtn.addEventListener('click', () => {
  // Math.max → 0未満にならないよう下限を設定
  const newIndex = Math.max(currentIndex - 1, 0);
  scrollToSlide(newIndex);
});

nextBtn.addEventListener('click', () => {
  // Math.min → スライド数を超えないよう上限を設定
  const newIndex = Math.min(currentIndex + 1, slides.length - 1);
  scrollToSlide(newIndex);
});

// --- ドットのクリックイベント ---
dots.forEach((dot, index) => {
  dot.addEventListener('click', () => {
    scrollToSlide(index);
  });
});

// --- キーボードナビゲーション ---
// カルーセルにフォーカスがある時、左右矢印キーで操作可能にする
track.addEventListener('keydown', (e) => {
  if (e.key === 'ArrowLeft') {
    // ArrowLeft → 前のスライドへ
    e.preventDefault(); // デフォルトのスクロール動作を防ぐ
    const newIndex = Math.max(currentIndex - 1, 0);
    scrollToSlide(newIndex);
  } else if (e.key === 'ArrowRight') {
    // ArrowRight → 次のスライドへ
    e.preventDefault();
    const newIndex = Math.min(currentIndex + 1, slides.length - 1);
    scrollToSlide(newIndex);
  }
});
<!-- role="region" → スクリーンリーダーにランドマーク領域として認識させる -->
<!-- aria-label → この領域の名前を設定（読み上げに使われる） -->
<!-- aria-roledescription → 「カルーセル」と読み上げさせる -->
<div class="carousel"
     role="region"
     aria-label="画像カルーセル"
     aria-roledescription="カルーセル">

  <!-- スクロールトラック：スライドが横に並ぶコンテナ -->
  <div class="carousel__track" id="carouselTrack">

    <!-- role="group" → 関連する要素のグループとして認識させる -->
    <!-- aria-roledescription="スライド" → 各グループを「スライド」と読み上げ -->
    <!-- aria-label → 現在位置を「スライド 1 / 5」のように伝える -->
    <div class="carousel__slide"
         role="group"
         aria-roledescription="スライド"
         aria-label="スライド 1 / 5">
      スライド 1
    </div>

    <div class="carousel__slide"
         role="group"
         aria-roledescription="スライド"
         aria-label="スライド 2 / 5">
      スライド 2
    </div>

    <!-- ... 残りのスライドも同様 ... -->
  </div>

  <!-- 矢印ボタン：aria-labelで操作内容を伝える -->
  <button class="carousel__arrow carousel__arrow--prev"
          aria-label="前のスライド">
    &#10094;
  </button>
  <button class="carousel__arrow carousel__arrow--next"
          aria-label="次のスライド">
    &#10095;
  </button>

  <!-- ドットナビゲーション -->
  <!-- role="tablist" → タブのようなナビゲーション要素のグループ -->
  <div class="carousel__dots"
       role="tablist"
       aria-label="スライド選択">

    <!-- role="tab" → 各ドットをタブとして認識させる -->
    <!-- aria-selected → 現在選択されているかどうか -->
    <button class="carousel__dot is-active"
            role="tab"
            aria-selected="true"
            aria-label="スライド1へ移動"></button>

    <button class="carousel__dot"
            role="tab"
            aria-selected="false"
            aria-label="スライド2へ移動"></button>

    <!-- ... 残りのドットも同様 ... -->
  </div>
</div>

実運用でのエッジケース

スライド1枚のみ（矢印・ドット不要のケース）

スライドが1枚しかない場合は、矢印ボタンとドットを非表示にするべきです。CSSで .carousel__dots:has(.carousel__dot:only-child) や、JSでスライド数を確認して制御します。

画像読み込み失敗時のフォールバック

実際の画像を使う場合、読み込みエラーに備えて背景色やプレースホルダー画像を設定しておくと、壊れた画像アイコンが表示されるのを防げます。

キーボード操作（左右矢印キー）

上のプレビューでカルーセルをクリックしてフォーカスし、左右矢印キーでスライドを移動してみてください。ドットが連動して更新されることを確認できます。
