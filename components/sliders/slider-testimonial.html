テスティモニアルスライダー

設計の意図

transform: translateX vs scroll-snap
画像カルーセルではscroll-snapを使いましたが、テスティモニアルスライダーではtransform: translateXを使います。translateXはアニメーションの制御がより細かく、自動再生やスワイプとの連携がしやすい利点があります。GPUアクセラレーションが効くためパフォーマンスも良好です。

自動再生には必ず一時停止ボタンを
WCAG 2.2.2では、自動で動くコンテンツにはユーザーが一時停止できる手段を提供することが求められています。また、ホバー中は自動再生を止めることで、読んでいる最中にスライドが変わるストレスを防ぎます。

タッチスワイプの実装
touchstart/touchmoveイベントでスワイプ方向を検出します。ライブラリなしで実装でき、50px以上の移動でスライド切り替えを発火させることで、意図しない切り替えを防ぎます。

/* --- スライダーコンテナ --- */
.testimonial-slider {
  position: relative;
  max-width: 560px;
  /* overflow: hidden → はみ出したスライドを隠す */
  overflow: hidden;
  border-radius: 12px;
}

/* --- トラック: 全スライドが横一列に並ぶ --- */
.testimonial-track {
  display: flex;
  /* transition → transform の変化をなめらかにアニメーション。
     cubic-bezier で加速→減速のイージングを指定 */
  transition: transform 500ms cubic-bezier(0.25, 0.46, 0.45, 0.94);
  /* will-change → ブラウザにアニメーション予定を伝え、
     GPUレイヤーを事前に準備させる最適化ヒント */
  will-change: transform;
}

/* --- 各スライド --- */
.testimonial-slide {
  /* flex: 0 0 100% → 各スライドが親の幅いっぱいに固定 */
  flex: 0 0 100%;
  padding: 32px 24px;
  text-align: center;
  /* user-select: none → スワイプ時のテキスト選択を防止 */
  user-select: none;
}

/* --- イニシャルアバター --- */

  /* グラデーション背景でバリエーションを作る */
  background: linear-gradient(135deg, #667eea, #764ba2);

const track = document.getElementById('testimonialTrack');
const slides = track.querySelectorAll('.testimonial-slide');
const dots = document.querySelectorAll('.testimonial-dot');
let currentIndex = 0;
let autoPlayId = null;
let isPaused = false;

// --- スライドを指定インデックスに移動 ---
function goToSlide(index) {
  // スライド数でループ（0未満→最後、最大超→最初）
  currentIndex = (index + slides.length) % slides.length;

  // translateX でトラック全体を横にずらす
  // -100% × インデックス で目的のスライドを表示
  track.style.transform =
    'translateX(-' + (currentIndex * 100) + '%)';

  // ドットの更新
  dots.forEach((dot, i) => {
    dot.classList.toggle('is-active', i === currentIndex);
  });
}

// --- 自動再生 ---
function startAutoPlay() {
  // setInterval → 指定ミリ秒ごとに関数を繰り返し実行
  autoPlayId = setInterval(() => goToSlide(currentIndex + 1), 4000);
}

function stopAutoPlay() {
  // clearInterval → setIntervalの繰り返しを停止
  clearInterval(autoPlayId);
}

// --- 一時停止ボタン ---
pauseBtn.addEventListener('click', () => {
  isPaused = !isPaused;
  if (isPaused) {
    stopAutoPlay();
    pauseBtn.textContent = '▶';
    pauseBtn.setAttribute('aria-label', '自動再生を再開');
  } else {
    startAutoPlay();
    pauseBtn.textContent = '⏸';
    pauseBtn.setAttribute('aria-label', '自動再生を一時停止');
  }
});

// --- ホバーで一時停止（読んでいる最中の切り替えを防ぐ） ---
slider.addEventListener('mouseenter', () => {
  if (!isPaused) stopAutoPlay();
});
slider.addEventListener('mouseleave', () => {
  if (!isPaused) startAutoPlay();
});

// --- タッチスワイプ ---
let touchStartX = 0;
track.addEventListener('touchstart', (e) => {
  // touches[0].clientX → 最初の指のX座標
  touchStartX = e.touches[0].clientX;
});
track.addEventListener('touchend', (e) => {
  const diff = touchStartX - e.changedTouches[0].clientX;
  // 50px以上の移動でスライド切り替え（誤操作防止）
  if (Math.abs(diff) > 50) {
    goToSlide(currentIndex + (diff > 0 ? 1 : -1));
  }
});

// 自動再生を開始
startAutoPlay();

実運用でのエッジケース

長いテスティモニアルテキスト

テキスト量がスライドごとに異なる場合、スライダーの高さが変動します。最も長いスライドに合わせて高さを固定するか、min-heightを設定してレイアウトの揺れを防ぎます。

自動再生の一時停止テスト

右上の⏸ボタンをクリックして自動再生を停止/再開してみてください。また、スライダーにマウスを乗せると自動再生が一時停止することも確認できます。

タッチスワイプ（モバイル）

モバイルデバイスまたはDevToolsのデバイスシミュレーターで、左右にスワイプしてスライドが切り替わることを確認してください。
