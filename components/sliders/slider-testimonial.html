設計の意図

transformベース vs scroll-snap
scroll-snapはスクロール可能なコンテンツに最適ですが、自動再生やスワイプ閾値の細かい制御が必要な場合はtransformベースが適しています。テスティモニアルのように1枚ずつ見せたい場合に向いています。

タッチジェスチャーの実装
touchstart/touchmove/touchendイベントで指の移動量を計算し、一定の閾値（50px）を超えたらスライド切り替えを実行します。短い距離のタッチは誤操作として無視します。

自動再生と一時停止
自動再生はユーザーの注意を引きますが、読んでいる途中で切り替わるとストレスになります。ホバー時に一時停止することでこの問題を解決します。

ソーシャルプルーフ
テスティモニアルは「社会的証明」の心理効果を利用するUIパターンです。実名・肩書き・アバターを添えることで信頼性が高まります。

コード解説

.testimonial-slider {
  /* overflow: hidden → はみ出したスライドを隠す */
  overflow: hidden;
  /* position: relative → 矢印ボタンの配置基準 */
  position: relative;
}

.testimonial-track {
  /* display: flex → スライドを横一列に並べる */
  display: flex;
  /* transition → translateXの変化をなめらかにアニメーション */
  transition: transform 500ms cubic-bezier(0.16, 1, 0.3, 1);
}

.testimonial-slide {
  /* min-width: 100% → 各スライドが親と同じ幅（1枚分） */
  min-width: 100%;
  /* flexの子要素はデフォルトで縮むため、min-widthで防ぐ */
}

/* 引用符の装飾 */
.testimonial-quote::before {
  /* content: "\201C" → 左ダブル引用符（"）のUnicodeコード */
  content: "\201C";
  position: absolute;
  font-size: 48px;
  /* 薄い色で控えめに装飾（内容の邪魔をしない） */
  color: var(--color-brand-200);
}
var track = document.getElementById('tsTrack');
var slides = track.children;
var current = 0;
var autoplayId = null;

/* スライドを指定インデックスに移動 */
function goTo(index) {
  /* スライド数を超えないように制限 */
  if (index < 0) index = slides.length - 1;
  if (index >= slides.length) index = 0;
  current = index;
  /* translateX(-N00%) → N枚目分だけ左にずらす */
  track.style.transform = 'translateX(-' + (current * 100) + '%)';
  /* ドットの更新 */
  updateDots();
}

/* 自動再生: 4秒ごとに次のスライドへ */
function startAutoplay() {
  autoplayId = setInterval(function() { goTo(current + 1); }, 4000);
}
function stopAutoplay() {
  clearInterval(autoplayId);
  autoplayId = null;
}

/* タッチスワイプ */
var startX = 0;
track.addEventListener('touchstart', function(e) {
  startX = e.touches[0].clientX;
  stopAutoplay(); /* 操作中は自動再生を止める */
});
track.addEventListener('touchend', function(e) {
  var diff = startX - e.changedTouches[0].clientX;
  /* 50px以上の移動でスライド切り替え（誤操作防止） */
  if (Math.abs(diff) > 50) {
    goTo(current + (diff > 0 ? 1 : -1));
  }
  startAutoplay();
});

/* ホバーで一時停止 */
slider.addEventListener('mouseenter', stopAutoplay);
slider.addEventListener('mouseleave', startAutoplay);
