スライダー・カルーセル

レンジスライダー

数値の選択や範囲の絞り込みに使うスライダーコンポーネント。ネイティブの <input type="range"> をカスタマイズし、視覚的なフィードバックとクロスブラウザ対応を実現します。

0 100

価格帯 ¥1,000 — ¥8,000
¥0 ¥10,000

設計の意図

なぜネイティブ range input をカスタマイズするのか？
ブラウザ標準の <input type="range"> は、Chrome・Firefox・Safari で見た目が大きく異なります。appearance: none でデフォルトスタイルをリセットし、ベンダープレフィックス付きの擬似要素（::-webkit-slider-thumb 等）で統一的なデザインを実現します。

デュアルレンジで価格・範囲フィルタリング
ECサイトの価格フィルターや、不動産の面積検索など「最小値〜最大値」を同時に指定する場面で使います。2つの <input type="range"> を重ね、JavaScript で最小値が最大値を超えないように制御します。

視覚的フィードバックの重要性
選択中の値を追従するツールチップや、塗りつぶされたトラック（フィルドトラック）は、現在の選択値を直感的に伝えます。数値だけでなく「割合」を色で表すことで、ユーザーは素早く現在地を把握できます。

モバイルでのタッチターゲット
つまみ（thumb）のサイズは最低 44px × 44px を確保するのが推奨です。見た目の円は 22px でも、::-webkit-slider-thumb のタッチ領域は十分な大きさが必要で、指で正確に操作できるようにします。

コード解説

/* === シングルスライダーのトラック・つまみ === */

.range-input {
  /* appearance: none → ブラウザ標準の見た目をすべてリセットする。
     これがないとカスタムスタイルが効かない */
  -webkit-appearance: none;
  appearance: none;

  /* width: 100% → 親要素の横幅いっぱいに広げる */
  width: 100%;

  /* height: 6px → トラック（レール）の太さを指定する */
  height: 6px;

  /* border-radius: 3px → トラックの角を丸くして柔らかい印象にする */
  border-radius: 3px;

  /* background → トラックの背景色。
     JSからlinear-gradientで塗り分ける（左：ブランド色、右：グレー） */
  background: var(--color-neutral-200);

  /* outline: none → フォーカス時のデフォルト枠を消す（別途 focus-visible で対応） */
  outline: none;

  /* cursor: pointer → つまみ以外のトラック部分もクリックで移動できることを示す */
  cursor: pointer;
}

/* ::-webkit-slider-thumb → Chrome/Safari/Edgeのつまみ */
.range-input::-webkit-slider-thumb {
  -webkit-appearance: none;
  appearance: none;

  /* width/height: 22px → つまみの直径。タッチしやすいサイズ */
  width: 22px;
  height: 22px;

  /* border-radius: 50% → 円形にする */
  border-radius: 50%;

  /* background → つまみの塗り色。ブランドカラーで目立たせる */
  background: var(--color-brand-500);

  /* border: 3px solid #fff → 白い縁取りでトラックとの境界を明確にする */
  border: 3px solid #fff;

  /* box-shadow → 影で浮き上がった感覚を出す */
  box-shadow: 0 1px 4px rgba(0, 0, 0, 0.25);

  /* margin-top: -8px → つまみをトラックの中央に配置する調整値
     計算: -(つまみ高さ - トラック高さ) / 2 = -(22-6)/2 = -8 */
  margin-top: -8px;

  cursor: pointer;
}

/* ::-webkit-slider-thumb:hover → ホバー時に光の輪を出す */
.range-input::-webkit-slider-thumb:hover {
  /* 薄いブランドカラーの輪で「触れている」フィードバック */
  box-shadow: 0 0 0 6px hsla(222, 80%, 50%, 0.15),
              0 1px 4px rgba(0, 0, 0, 0.25);
}

/* ::-moz-range-thumb → Firefoxのつまみ */
.range-input::-moz-range-thumb {
  width: 16px;
  height: 16px;
  border-radius: 50%;
  background: var(--color-brand-500);
  border: 3px solid #fff;
  box-shadow: 0 1px 4px rgba(0, 0, 0, 0.25);
  cursor: pointer;
}

/* ::-moz-range-track → Firefoxのトラック */
.range-input::-moz-range-track {
  /* height: 6px → トラックの太さ */
  height: 6px;
  border-radius: 3px;
  background: var(--color-neutral-200);
}

/* === フィルドトラック（塗りつぶし）のテクニック === */
/* linear-gradientでトラック背景を動的に塗り分ける */
/* JSから以下のように設定する：
   background: linear-gradient(
     to right,
     var(--color-brand-500) 0%,
     var(--color-brand-500) 50%,  ← 現在値の割合
     var(--color-neutral-200) 50%,
     var(--color-neutral-200) 100%

/* === デュアルスライダーの塗り部分 === */
.dual-track-fill {
  /* position: absolute → 親の .dual-track 内で自由に配置する */
  position: absolute;
  top: 0;

  /* height: 100% → トラックと同じ高さにする */
  height: 100%;

  /* background → min〜max間を塗りつぶすブランドカラー */
  background: var(--color-brand-500);
  border-radius: 3px;

  /* left と width は JS から動的に設定する */
}

.dual-slider-wrap input[type="range"] {
  -webkit-appearance: none;
  appearance: none;
  position: absolute;
  top: 0;
  left: 0;
  width: 100%;
  height: 6px;

  /* background: transparent → トラック背景を透明にして

  background: transparent;

  /* pointer-events: none → トラック部分のクリックを無効化し
     つまみだけ操作可能にする（つまみ側でautoに戻す） */

.dual-slider-wrap input[type="range"]::-webkit-slider-thumb {
  /* pointer-events: auto → つまみだけクリック可能にする */
  pointer-events: auto;

  /* z-index: 2 → つまみを最前面に出して確実に掴めるようにする */
  z-index: 2;

/* === シングルスライダーのロジック === */

// 要素を取得する
const slider = document.getElementById('singleSlider');
const valueLabel = document.getElementById('singleValue');

// updateSingle() → スライダーの値と見た目を同期する関数
function updateSingle() {
  const val = Number(slider.value);   // 現在値を数値として取得
  const min = Number(slider.min);     // 最小値
  const max = Number(slider.max);     // 最大値

  // percent → 現在値が全体の何%かを計算する
  const percent = ((val - min) / (max - min)) * 100;

  // ツールチップのテキストを更新する
  valueLabel.textContent = val;

  // ツールチップの位置をつまみに追従させる
  // left: percent% で大まかな位置を決め、
  // つまみの幅によるズレを補正する（中央で0px、端で±11px）
  const thumbOffset = 11 - (percent / 100) * 22;
  valueLabel.style.left = `calc(${percent}% + ${thumbOffset}px)`;

  // フィルドトラック → linear-gradientで左側をブランド色に塗る
  slider.style.background =
    `linear-gradient(to right, ` +
    `var(--color-brand-500) 0%, ` +
    `var(--color-brand-500) ${percent}%, ` +
    `var(--color-neutral-200) ${percent}%, ` +
    `var(--color-neutral-200) 100%)`;
}

// input イベント → つまみをドラッグするたびに発火する
slider.addEventListener('input', updateSingle);

// 初期表示でも値を反映する
updateSingle();


/* === デュアルスライダーのロジック === */

const minInput = document.getElementById('dualMin');
const maxInput = document.getElementById('dualMax');
const fill = document.getElementById('dualFill');
const display = document.getElementById('dualDisplay');

// formatYen() → 数値を日本円表記にフォーマットする関数
function formatYen(num) {
  // toLocaleString('ja-JP') → 3桁ごとにカンマを入れる
  return '¥' + Number(num).toLocaleString('ja-JP');
}

// updateDual() → デュアルスライダーの値と見た目を同期する関数
function updateDual() {
  let minVal = Number(minInput.value);
  let maxVal = Number(maxInput.value);
  const rangeMin = Number(minInput.min);
  const rangeMax = Number(minInput.max);

  // クランプ処理 → 最小値が最大値を超えないようにする
  if (minVal > maxVal) {
    // どちらの入力が最後に動いたかで、超えた方を戻す
    // ここでは minInput を maxVal に合わせる
    minInput.value = maxVal;
    minVal = maxVal;
  }
  if (maxVal < minVal) {
    maxInput.value = minVal;
    maxVal = minVal;
  }

  // パーセンテージ位置を計算する
  const minPercent = ((minVal - rangeMin) / (rangeMax - rangeMin)) * 100;
  const maxPercent = ((maxVal - rangeMin) / (rangeMax - rangeMin)) * 100;

  // fill要素のleftとwidthを更新して、選択範囲を視覚的に示す
  fill.style.left = minPercent + '%';
  fill.style.width = (maxPercent - minPercent) + '%';

  // 価格表示を更新する
  display.textContent = formatYen(minVal) + ' — ' + formatYen(maxVal);
}

// 両方のinputにイベントリスナーを設定する
minInput.addEventListener('input', function () {
  // 最小値が最大値を超えた場合、最大値に制限する
  if (Number(minInput.value) > Number(maxInput.value)) {
    minInput.value = maxInput.value;
  }
  updateDual();
});

maxInput.addEventListener('input', function () {
  // 最大値が最小値を下回った場合、最小値に制限する
  if (Number(maxInput.value) < Number(minInput.value)) {
    maxInput.value = minInput.value;
  }
  updateDual();
});

// 初期表示でも値を反映する
updateDual();
