ナビゲーション

ページネーション

コンテンツを複数ページに分割し、ユーザーが任意のページへ移動できるナビゲーション。無限スクロールや「もっと読む」ボタンと違い、ユーザーが自分の位置を常に把握できる予測可能なUIです。

プレビュー

ページをクリックして操作できます（全10ページ）

実運用でのエッジケース

1ページのみ（ページネーション非表示）

→ 1ページしかない場合、ページネーション自体を非表示にする

3ページ（省略記号なし）
100ページ（大量ページ）

設計の意図

ページネーション vs 無限スクロール vs 「もっと読む」ボタン
ページネーションは、ユーザーが自分の位置を常に把握できるため、検索結果や管理画面の一覧表示に適しています。無限スクロールはSNSフィードのような「流し見」に向き、「もっと読む」はその中間です。ページネーションの強みは「3ページ目の結果をもう一度見たい」という再訪が容易な点です。

先頭・末尾ページを常に表示する理由
最初と最後のページ番号を常に見せることで、全体のボリューム感をユーザーに伝えます。「100ページ中の50ページ目にいる」という文脈情報が、現在位置の理解を助けます。

省略記号（...）の役割
すべてのページ番号を表示すると、100ページの場合は100個のボタンが並んでしまいます。省略記号で間を省略し、現在ページの前後と先頭・末尾だけを表示することで、コンパクトかつ実用的なUIを保ちます。

コード解説

<!-- nav要素 → ナビゲーション用のランドマーク。
     aria-label でスクリーンリーダーに「これはページネーションです」と伝える -->
<nav class="pagination" aria-label="ページネーション">

  <!-- 「前へ」ボタン → 1ページ目のときは disabled にする
       aria-label で「前のページへ」と明示し、「<」だけでも意味が伝わるようにする -->
  <button class="pagination__btn pagination__btn--prev pagination__btn--disabled"
          aria-label="前のページへ" disabled>
    &lt; 前へ
  </button>

  <!-- ページ番号ボタン → aria-label="ページ 1" で各ボタンの意味を明示 -->
  <button class="pagination__btn pagination__btn--active"
          aria-current="page" aria-label="ページ 1">1</button>

  <button class="pagination__btn" aria-label="ページ 2">2</button>
  <button class="pagination__btn" aria-label="ページ 3">3</button>

  <!-- 省略記号 → span で表示。ボタンではないのでクリック不可 -->
  <span class="pagination__ellipsis" aria-hidden="true">&hellip;</span>

  <button class="pagination__btn" aria-label="ページ 10">10</button>

  <!-- 「次へ」ボタン → 最終ページでは disabled にする -->
  <button class="pagination__btn pagination__btn--next"
          aria-label="次のページへ">
    次へ &gt;
  </button>
</nav>
.pagination {
  /* display: flex → ボタンを横一列に並べる。
     Flexboxはこのような「一次元の並び」に最適 */
  display: flex;
  align-items: center;

  /* gap: 4px → ボタン間の余白。
     margin を各ボタンに付ける代わりに gap で一括管理する。
     4pxは隣同士が近すぎず離れすぎない絶妙な間隔 */
  gap: 4px;

  /* flex-wrap: wrap → 画面幅が狭い時に折り返す。
     モバイルでボタンがはみ出すのを防ぐ */
  flex-wrap: wrap;
}

.pagination__btn {
  /* min-width: 36px → 「1」でも「10」でも同じ最小幅を保つ。
     数字の桁数が変わってもボタンサイズがガタつかない */
  min-width: 36px;
  height: 36px;

  /* padding: 0 8px → 上下は height で確保済みなので 0。
     左右に余裕を持たせて「10」などの2桁数字も窮屈にならない */
  padding: 0 8px;

  /* border: 1px solid → 枠線ありのアウトラインスタイル。
     アクティブページの塗りつぶしと対比させることで、
     現在ページがどこか一目でわかる */
  border: 1px solid var(--color-border);
  border-radius: var(--radius-md);
  background: transparent;
  cursor: pointer;

  transition: background-color 150ms,
              color 150ms,
              border-color 150ms;
}

/* --- アクティブ（現在のページ） ---
   背景をブランドカラーで塗りつぶし、他のアウトラインボタンと差別化する。
   この「塗り vs 枠線のみ」のコントラストが視認性のカギ */
.pagination__btn--active {
  background-color: var(--color-brand-500);
  border-color: var(--color-brand-500);
  color: white;
  font-weight: 600;
  cursor: default;
}

/* --- 無効状態（最初/最後のページで前へ/次へを無効化） ---
   pointer-events: none → CSSレベルでクリックを完全に無効化する。
   HTML の disabled 属性と組み合わせることで二重にガードする。
   opacity: 0.4 → 半透明にして「操作不可」を視覚的に伝える */
.pagination__btn--disabled {
  pointer-events: none;
  opacity: 0.4;
  cursor: not-allowed;
}

/* --- 省略記号（...）→ ボタンではなくテキスト表示 ---
   クリックできないことを明確にするため、
   button ではなく span で実装する */
.pagination__ellipsis {
  min-width: 36px;
  height: 36px;
  display: inline-flex;
  align-items: center;
  justify-content: center;
  color: var(--color-text-tertiary);
  cursor: default;
}
/**
 * ページネーションを生成する関数
 *
 * 省略記号（...）のロジック:
 * - 常に表示するもの: 最初のページ、最後のページ、現在ページの前後1ページ
 * - それ以外は省略記号で省く
 * - 例: 現在5ページ目 → [1] ... [4] [5] [6] ... [10]
 */
function generatePageNumbers(currentPage, totalPages) {
  const pages = [];

  // === ページ数が7以下なら全ページ表示（省略不要） ===
  if (totalPages <= 7) {
    for (let i = 1; i <= totalPages; i++) {
      pages.push(i);
    }
    return pages;
  }

  // === 常に最初のページを追加 ===
  pages.push(1);

  // === 現在ページが最初から離れていたら省略記号を挿入 ===
  // currentPage > 3 なら [1] の次に ... が必要
  if (currentPage > 3) {
    pages.push('...');
  }

  // === 現在ページの前後1ページを追加（隣接ページ） ===
  // Math.max/min で配列の範囲が 2〜(totalPages-1) に収まるようにする
  const start = Math.max(2, currentPage - 1);
  const end = Math.min(totalPages - 1, currentPage + 1);

  for (let i = start; i <= end; i++) {
    pages.push(i);
  }

  // === 現在ページが最後から離れていたら省略記号を挿入 ===
  if (currentPage < totalPages - 2) {
    pages.push('...');
  }

  // === 常に最後のページを追加 ===
  pages.push(totalPages);

  return pages;
}

/**
 * ページネーションのHTMLを描画する
 *
 * @param {HTMLElement} container - ページネーションを入れるnav要素
 * @param {number} currentPage - 現在のページ番号
 * @param {number} totalPages - 総ページ数
 * @param {function} onPageChange - ページ変更時のコールバック関数
 */
function renderPagination(container, currentPage, totalPages, onPageChange) {
  // === 1ページのみの場合はページネーション自体を非表示 ===
  if (totalPages <= 1) {
    container.style.display = 'none';
    return;
  }
  container.style.display = 'flex';

  // === HTMLを組み立てる ===
  let html = '';

  // --- 「前へ」ボタン ---
  // 1ページ目なら disabled クラスと disabled 属性を付与
  const prevDisabled = currentPage === 1;
  html += `<button class="pagination__btn pagination__btn--prev${
    prevDisabled ? ' pagination__btn--disabled' : ''
  }" aria-label="前のページへ"${
    prevDisabled ? ' disabled' : ''
  }>&lt; 前へ</button>`;

  // --- ページ番号 ---
  const pages = generatePageNumbers(currentPage, totalPages);

  pages.forEach(page => {
    if (page === '...') {
      // 省略記号 → aria-hidden="true" でスクリーンリーダーから隠す
      html += '<span class="pagination__ellipsis" aria-hidden="true">&hellip;</span>';
    } else {
      // ページ番号ボタン → 現在ページには aria-current="page" を付与
      const isActive = page === currentPage;
      html += `<button class="pagination__btn${
        isActive ? ' pagination__btn--active' : ''
      }" aria-label="ページ ${page}"${
        isActive ? ' aria-current="page"' : ''
      }>${page}</button>`;
    }
  });

  // --- 「次へ」ボタン ---
  const nextDisabled = currentPage === totalPages;
  html += `<button class="pagination__btn pagination__btn--next${
    nextDisabled ? ' pagination__btn--disabled' : ''
  }" aria-label="次のページへ"${
    nextDisabled ? ' disabled' : ''
  }>次へ &gt;</button>`;

  container.innerHTML = html;

  // === クリックイベントを設定 ===
  container.querySelectorAll('.pagination__btn').forEach(btn => {
    btn.addEventListener('click', () => {
      if (btn.disabled) return;

      if (btn.classList.contains('pagination__btn--prev')) {
        onPageChange(currentPage - 1);
      } else if (btn.classList.contains('pagination__btn--next')) {
        onPageChange(currentPage + 1);
      } else if (!btn.classList.contains('pagination__btn--active')) {
        // ページ番号ボタン → aria-label から番号を取得
        const pageNum = parseInt(btn.textContent, 10);
        onPageChange(pageNum);
      }
    });
  });
}
