ソート可能テーブル

ヘッダーをクリックして列の昇順・降順を切り替えられるテーブル。aria-sort属性で現在のソート状態をスクリーンリーダーに伝えます。

設計の意図

aria-sort でソート状態を伝える
aria-sort 属性は "none"（未ソート）、"ascending"（昇順）、"descending"（降順）の3状態を持ちます。視覚的な矢印アイコンと一緒に使うことで、目が見えるユーザーとスクリーンリーダーユーザーの双方に現在の状態を伝えます。

Array.sort() によるクライアントサイドソート
少量のデータであればJavaScriptでDOMを直接操作してソートできます。大量のデータ（数百行以上）ではサーバーサイドソートやvirtual scrollの検討が必要です。

数値列は右揃え
数値は桁を揃えて比較しやすくするため、右揃え（text-align: right）にするのが慣例です。テキスト列は左揃えのままにします。

コード解説

<table class="sortable-table">
  <thead>
    <tr>
      <!-- aria-sort="none" → まだソートされていない列 -->
      <!-- data-type → ソート時の比較方法を指定 -->
      <th aria-sort="none" data-type="string">名前</th>
      <th aria-sort="none" data-type="number">売上</th>
      <th aria-sort="none" data-type="date">入社日</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <td>田中太郎</td>
      <td class="text-right">1,250</td>
      <td>2020-04-01</td>
    </tr>
    <!-- ... -->
  </tbody>
</table>
/* ソート可能ヘッダーのカーソルとパディング */
.sortable-table th[aria-sort] {
  cursor: pointer;
  padding-right: 32px; /* 矢印分の余白 */
}

/* --- ソートインジケーター（::afterで矢印を表示） --- */
/* 未ソート：上下矢印を薄く表示 */
.sortable-table th[aria-sort]::after {
  content: '⇅';
  position: absolute;
  right: 12px;
  color: var(--color-neutral-300);
}

/* 昇順：上矢印をブランドカラーで強調 */
.sortable-table th[aria-sort="ascending"]::after {
  content: '↑';
  color: var(--color-brand-600);
}

/* 降順：下矢印をブランドカラーで強調 */
.sortable-table th[aria-sort="descending"]::after {
  content: '↓';
  color: var(--color-brand-600);
}
function initSortableTable(table) {
  const headers = table.querySelectorAll('th[aria-sort]');
  const tbody = table.querySelector('tbody');

  headers.forEach((header, colIndex) => {
    header.addEventListener('click', () => {
      // 現在のソート方向を取得し、次の方向を決定
      const currentSort = header.getAttribute('aria-sort');
      const nextSort = currentSort === 'ascending'
        ? 'descending' : 'ascending';

      // 全ヘッダーのaria-sortをリセット
      headers.forEach(h => h.setAttribute('aria-sort', 'none'));
      header.setAttribute('aria-sort', nextSort);

      // tbody内の行を配列に変換してソート
      const rows = [...tbody.querySelectorAll('tr')];
      const type = header.dataset.type; // string / number / date

      rows.sort((a, b) => {
        let aVal = a.cells[colIndex].textContent.trim();
        let bVal = b.cells[colIndex].textContent.trim();

        // データ型に応じた比較
        if (type === 'number') {
          // カンマを除去して数値化
          aVal = parseFloat(aVal.replace(/,/g, ''));
          bVal = parseFloat(bVal.replace(/,/g, ''));
        } else if (type === 'date') {
          aVal = new Date(aVal);
          bVal = new Date(bVal);
        }

        // 昇順/降順の切り替え
        const direction = nextSort === 'ascending' ? 1 : -1;

        if (type === 'string') {
          return direction * aVal.localeCompare(bVal, 'ja');
        }
        return direction * (aVal - bVal);
      });

      // ソート結果をDOMに反映
      rows.forEach(row => tbody.appendChild(row));
    });
  });
}
