ポップオーバー

クリックで表示されるフローティングUIパネル。プロフィールカードや3点メニューのドロップダウンなど、追加情報やアクションを提示するパターンです。外側クリックで閉じ、aria-expanded で開閉状態を伝えます。

設計の意図

クリックトリガー vs ホバートリガー
ツールチップはホバーで表示しますが、ポップオーバーはクリックで表示します。ポップオーバーにはボタンやリンクなどのインタラクティブ要素が含まれるため、ホバーでは操作しづらく、タッチデバイスでは動作しません。

外側クリックで閉じる理由
ユーザーがポップオーバーの外をクリックした時に閉じることは、「意図しない操作を防ぐ」ためです。Escapeキーでも閉じられるようにすると、キーボードユーザーの体験も向上します。

scale(0.95)→scale(1) のアニメーション
ポップオーバーが「ボタンから生まれる」ような印象を与えるために、小さい状態から実寸に拡大します。transform-originをトリガーの近くに設定することで、視覚的な連続性が生まれます。

コード解説

<!-- ポップオーバーのラッパー: position: relative の基準 -->
<div class="popover-wrapper">

  <!-- aria-expanded → 開閉状態をスクリーンリーダーに伝える -->
  <!-- aria-haspopup → この要素がポップアップを持つことを宣言 -->
  <button class="profile-trigger"
          aria-expanded="false"
          aria-haspopup="true">
    田中 太郎
  </button>

  <!-- role="dialog" → ポップオーバーをダイアログとして認識させる -->
  <div class="popover-content profile-card"
       role="dialog"
       aria-label="プロフィール">
    <!-- カードの中身 -->
  </div>
</div>

<!-- === ドロップダウンメニュー === -->
<div class="popover-wrapper">
  <button class="menu-trigger"
          aria-expanded="false"
          aria-haspopup="true"
          aria-label="操作メニュー">
    ⋮
  </button>

  <!-- role="menu" → メニュー項目のリストとして認識 -->
  <div class="popover-content dropdown-menu"
       role="menu"
       aria-label="操作メニュー">

    <!-- role="menuitem" → メニュー内の各項目 -->
    <button class="dropdown-item" role="menuitem">
      編集する
    </button>
    <button class="dropdown-item" role="menuitem">
      コピーする
    </button>

    <!-- 区切り線 -->
    <div class="dropdown-divider"></div>

    <!-- 危険な操作は色を変えて注意を促す -->
    <button class="dropdown-item dropdown-item--danger" role="menuitem">
      削除する
    </button>
  </div>
</div>
/* --- ポップオーバーの基盤スタイル --- */
.popover-wrapper {
  /* position: relative → 子要素のabsolute配置の基準点にする */
  position: relative;
  display: inline-block;
}

.popover-content {
  /* position: absolute → トリガーボタンからの相対位置で配置 */
  position: absolute;
  z-index: 10;

  /* 初期状態: 非表示 */
  opacity: 0;
  transform: scale(0.95);  /* わずかに小さい状態から始める */

  /* transform-origin → アニメーションの起点。
     トリガーに近い角を起点にすると自然な動きになる */
  transform-origin: top left;

  /* pointer-events: none → 非表示時はクリックを透過させる */
  pointer-events: none;

  /* opacity と transform をなめらかに変化させる */
  transition: opacity 150ms ease,
              transform 150ms ease;
}

/* is-open クラスで表示状態にする */
.popover-content.is-open {
  opacity: 1;
  transform: scale(1);       /* 実寸に戻す */
  pointer-events: auto;      /* クリックを受け付ける */
}

/* --- プロフィールカード --- */
.profile-card {
  /* トリガーの下に8pxの隙間を空けて配置 */
  top: calc(100% + 8px);
  left: 0;
  width: 280px;
  background: white;
  border: 1px solid #e2e8f0;
  border-radius: 12px;
  /* 2重シャドウで深みを出す（大きい影 + 小さい影） */
  box-shadow: 0 10px 25px rgba(0,0,0,0.1),
              0 4px 10px rgba(0,0,0,0.05);
}

/* --- ドロップダウンメニュー --- */
.dropdown-menu {
  top: calc(100% + 4px);
  right: 0;              /* 右揃え（3点ボタンの右端に合わせる） */
  left: auto;
  transform-origin: top right;  /* 右上を起点にアニメーション */
  min-width: 180px;
}

/* メニュー項目 */
.dropdown-item {
  width: 100%;
  padding: 8px 12px;
  border: none;
  background: none;
  text-align: left;
  cursor: pointer;
  border-radius: 4px;
  transition: background-color 150ms;
}

.dropdown-item:hover {
  background-color: #f7fafc;
}

/* 危険な操作は赤文字で警告 */
.dropdown-item--danger {
  color: #ef4444;
}

.dropdown-item--danger:hover {
  background-color: #fef2f2;  /* 薄い赤背景 */
}
// --- ポップオーバーの汎用制御クラス ---
function setupPopover(triggerId, contentId) {
  const trigger = document.getElementById(triggerId);
  const content = document.getElementById(contentId);

  // --- クリックで開閉をトグル ---
  trigger.addEventListener('click', (e) => {
    e.stopPropagation(); // イベントの伝播を止める
    const isOpen = content.classList.contains('is-open');

    // まず全てのポップオーバーを閉じる
    closeAllPopovers();

    if (!isOpen) {
      // 閉じていたら開く
      content.classList.add('is-open');
      trigger.setAttribute('aria-expanded', 'true');
    }
  });
}

// --- 全ポップオーバーを閉じる関数 ---
function closeAllPopovers() {
  document.querySelectorAll('.popover-content.is-open')
    .forEach(el => el.classList.remove('is-open'));
  document.querySelectorAll('[aria-expanded="true"]')
    .forEach(el => el.setAttribute('aria-expanded', 'false'));
}

// --- 外側クリックで閉じる ---
// document全体のクリックを監視
document.addEventListener('click', (e) => {
  // closest() → クリック位置から親方向に.popover-wrapperを探す
  // 見つからない = ポップオーバーの外側をクリックした
  if (!e.target.closest('.popover-wrapper')) {
    closeAllPopovers();
  }
});

// --- Escapeキーで閉じる ---
document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') {
    closeAllPopovers();
  }
});

// 各ポップオーバーを初期化
setupPopover('profileTrigger', 'profileCard');
setupPopover('menuTrigger', 'dropdownMenu');

実運用でのエッジケース

画面端での位置調整

ポップオーバーが画面の右端や下端からはみ出す場合、表示方向を自動で反転させるフリップ処理が必要です。Floating UIなどのライブラリを使うと効率的に実装できます。

複数ポップオーバーの排他制御

上のプレビューで両方のボタンを交互にクリックしてください。一方を開くともう一方が自動的に閉じることを確認できます。

キーボード操作（Escapeで閉じる）

ポップオーバーを開いた状態でEscapeキーを押すと閉じます。フォーカスがトリガーボタンに戻ることで、キーボードユーザーが迷子にならないようにします。
