設計の意図

なぜ未読インジケーターは青い丸なのか？
小さな青い丸は、最小限のスペースで「まだ読んでいない」ことを効果的に伝えます。テキストを太字にすることで、スキャン時にも未読メールが一目でわかります。

キーボードナビゲーションの重要性
リストはマウスだけでなく、ArrowUp/Downキーで移動できるべきです。role="listbox"とaria-selected属性により、スクリーンリーダーでも選択状態が正しく伝わります。

text-overflow: ellipsis でテキスト省略
件名が長い場合、折り返すと各アイテムの高さがバラバラになります。省略記号（...）で切ることで、一覧性を保ちつつ内容の手がかりを提供します。

コード解説

<!-- role="listbox" → 選択可能なリストであることをスクリーンリーダーに伝える -->
<ul class="inbox-list" role="listbox" aria-label="受信メール一覧">

  <!-- role="option" → リスト内の選択可能な項目 -->
  <!-- aria-selected → 選択状態をアクセシビリティツールに伝える -->
  <!-- data-unread → 未読/既読の状態をCSSで切り替えるためのカスタム属性 -->
  <!-- tabindex="0" → 最初のアイテムのみフォーカス可能にする -->
  <li class="inbox-item" role="option" aria-selected="false"
      data-unread="true" tabindex="0">

    <!-- 未読インジケーター（青い丸） -->
    <span class="inbox-unread-dot" aria-hidden="true"></span>

    <div class="inbox-content">
      <div class="inbox-sender">田中 太郎</div>
      <div class="inbox-subject">プロジェクトの進捗報告</div>
    </div>

    <span class="inbox-date">3月22日</span>
  </li>

  <!-- 2番目以降は tabindex="-1" → Tabでは飛ばし、矢印キーで移動 -->
  <li class="inbox-item" role="option" aria-selected="false"
      data-unread="false" tabindex="-1">
    ...
  </li>
</ul>
/* リスト全体のコンテナ */
.inbox-list {
  list-style: none;        /* デフォルトの黒丸を消す */
  border: 1px solid #e2e8f0;
  border-radius: 8px;
  overflow: hidden;        /* 角丸からはみ出さないようにする */
}

/* 各リストアイテム — grid で3列構成 */
.inbox-item {
  /* grid-template-columns → 未読丸(8px) / 内容(残り) / 日付(自動) */
  display: grid;
  grid-template-columns: 8px 1fr auto;
  align-items: center;
  gap: 12px;
  padding: 12px 16px;
  border-bottom: 1px solid #e2e8f0;
  cursor: pointer;
  transition: background-color 150ms;
  outline: none;           /* デフォルトのフォーカス枠を消す（独自スタイルで代替） */
}

/* ホバー時の背景色変更 — マウスが乗っていることを視覚的に伝える */
.inbox-item:hover {
  background-color: #f7fafc;
}

/* キーボードフォーカス時 — 左端に青い線を表示 */
.inbox-item:focus-visible {
  background-color: #f1f5f9;
  box-shadow: inset 3px 0 0 #3b82f6;  /* inset → 内側に影を描く */
}

/* 選択状態 — 薄い青背景で「選択中」を示す */
.inbox-item[aria-selected="true"] {
  background-color: #eff6ff;
}

/* 未読インジケーター — data属性で表示を切り替え */
.inbox-item[data-unread="true"] .inbox-unread-dot {
  background-color: #3b82f6;  /* 青い丸を表示 */
}

/* 未読アイテムのテキストを太字にして目立たせる */
.inbox-item[data-unread="true"] .inbox-sender,
.inbox-item[data-unread="true"] .inbox-subject {
  font-weight: 600;
}

/* 件名の省略表示 */
.inbox-subject {
  white-space: nowrap;       /* 折り返さない */
  overflow: hidden;          /* はみ出し部分を隠す */
  text-overflow: ellipsis;   /* 省略記号（...）を表示 */
}
// リスト要素と全アイテムを取得
const list = document.getElementById('inbox');
const items = [...list.querySelectorAll('.inbox-item')];

// --- クリックで選択をトグル ---
list.addEventListener('click', (e) => {
  // closest() → クリック位置から最も近い.inbox-itemを探す
  const item = e.target.closest('.inbox-item');
  if (!item) return;

  // 現在の選択状態を反転する
  const isSelected = item.getAttribute('aria-selected') === 'true';
  item.setAttribute('aria-selected', String(!isSelected));

  // フォーカスを移動
  setFocusToItem(item);
  updateSelectionCount();
});

// --- キーボードナビゲーション ---
list.addEventListener('keydown', (e) => {
  const current = document.activeElement;
  const idx = items.indexOf(current);
  if (idx === -1) return;

  let next;
  switch (e.key) {
    case 'ArrowDown':
      e.preventDefault();  // ページスクロールを防ぐ
      next = items[idx + 1] || items[0];  // 末尾→先頭にループ
      break;
    case 'ArrowUp':
      e.preventDefault();
      next = items[idx - 1] || items[items.length - 1];  // 先頭→末尾にループ
      break;
    case ' ':
    case 'Enter':
      e.preventDefault();
      current.click();  // スペース/Enterで選択トグル
      return;
    default:
      return;
  }

  setFocusToItem(next);
});

// --- ロービングタブインデックス（フォーカス管理） ---
function setFocusToItem(item) {
  // 全アイテムのtabindexを-1にリセット
  items.forEach(i => i.setAttribute('tabindex', '-1'));
  // 対象だけを0にしてフォーカスを当てる
  item.setAttribute('tabindex', '0');
  item.focus();
}

// --- 選択件数の表示を更新 ---
function updateSelectionCount() {
  const count = items.filter(
    i => i.getAttribute('aria-selected') === 'true'
  ).length;
  document.getElementById('selectionInfo').textContent =
    '選択中: ' + count + '件';
}
