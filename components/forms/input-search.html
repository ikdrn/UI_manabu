検索入力

検索入力は、ユーザーがキーワードを入力してリストやデータを絞り込むためのUIです。SVGアイコン、クリアボタン、ドロップダウン候補表示を組み合わせた実践的なパターンを学びます。

プレビュー

設計の意図

なぜ虫眼鏡アイコンを左に置くのか？
虫眼鏡は「検索」の世界共通アイコンです。入力欄の左端に配置することで、ユーザーはフィールドの用途を一目で理解できます。position: absoluteでアイコンを重ね、padding-leftでテキストと被らないようにします。

debounce（遅延実行）パターンの重要性
キー入力のたびにAPIリクエストを送ると、サーバーに過剰な負荷がかかります。debounceは「最後の入力から一定時間（例: 300ms）経ってから処理を実行する」パターンです。setTimeoutとclearTimeoutで実装します。

<mark>タグでマッチ箇所を強調する
検索結果の中で入力したキーワードに一致する部分をハイライトすると、ユーザーは結果の関連性をすばやく判断できます。HTMLの<mark>タグはセマンティックにも正しいマークアップです。

クリアボタンは入力がある時だけ表示する
空の状態でクリアボタンを見せても意味がありません。JavaScriptで入力の有無を監視し、display: noneとdisplay: inline-flexを切り替えます。

/* --- 検索ラッパー --- */
.search-wrapper {
  /* position: relative → 子要素のabsolute配置の基準点にする。
     アイコン・クリアボタン・ドロップダウンの位置がここを基準に決まる */
  position: relative;
  width: 100%;
  max-width: 400px;
}

/* --- 虫眼鏡アイコン --- */
.search-icon {
  /* position: absolute → 入力欄の上に重ねて配置する */
  position: absolute;
  left: 12px;
  top: 50%;
  /* translateY(-50%) → 垂直方向の中央揃え。
     top: 50%だけだとアイコン上端が中央になるため、
     自身の高さの半分だけ上にずらす */
  transform: translateY(-50%);
  /* pointer-events: none → アイコンのクリックを貫通させて、
     下のinputにフォーカスが届くようにする */
  pointer-events: none;
}

/* --- 検索入力 --- */
.search-input {
  /* padding: 10px 40px 10px 38px →
     左に38px → アイコン分の余白
     右に40px → クリアボタン分の余白 */
  padding: 10px 40px 10px 38px;
  /* border-radius: 9999px → ピル型（完全な丸角）にすると
     「検索フィールドらしさ」が出る */
  border-radius: 9999px;
}

/* --- ドロップダウン --- */
.search-results {
  /* position: absolute → 入力欄の真下に浮かせて表示する */
  position: absolute;
  top: calc(100% + 4px);
  left: 0;
  right: 0;
  /* box-shadow → 浮いている感覚を出す影 */
  box-shadow: 0 8px 24px hsla(220, 15%, 10%, 0.12);
  max-height: 240px;
  /* overflow-y: auto → 候補が多い時にスクロール可能にする */
  overflow-y: auto;
  /* z-index → 他の要素より手前に表示する */
  z-index: 100;
  display: none;
}

.search-results.open {
  display: block;
}

/* --- マッチ箇所ハイライト --- */
.search-result-item mark {
  /* <mark>タグ → HTMLが持つ「強調表示」専用のタグ。
     デフォルトは黄色背景だが、カスタマイズして使う */
  background-color: hsl(48, 100%, 75%);
  border-radius: 2px;

// --- 検索データ（実際にはAPIから取得することが多い） ---
const items = [
  '寿司', 'ラーメン', '天ぷら', 'うどん', 'そば',
  'カレーライス', 'とんかつ', '焼き鳥', 'お好み焼き',
  'たこ焼き', '刺身', '味噌汁', '餃子', '唐揚げ', '肉じゃが'
];

const input = document.getElementById('search-demo');
const results = document.getElementById('search-results-list');
const clearBtn = document.querySelector('.search-clear');
let debounceTimer;

// --- debounce（遅延実行）パターン ---
// キー入力のたびにタイマーをリセットし、
// 300ms入力が止まったら検索を実行する
input.addEventListener('input', () => {
  clearTimeout(debounceTimer);
  debounceTimer = setTimeout(() => {
    filterItems(input.value.trim());
  }, 300);

  // クリアボタンの表示切り替え
  clearBtn.classList.toggle('visible', input.value.length > 0);
});

// --- フィルタリング＆ハイライト ---
function filterItems(query) {
  if (!query) {
    results.classList.remove('open');
    return;
  }

  // filter() → 配列から条件に合う要素だけを抽出する
  const matched = items.filter(item =>
    item.includes(query)
  );

  if (matched.length === 0) {
    results.innerHTML =
      '<div class="search-no-result">一致する結果がありません</div>';
  } else {
    results.innerHTML = matched.map(item => {
      // マッチ箇所を<mark>で囲んでハイライトする
      const highlighted = item.replace(
        new RegExp(`(${escapeRegex(query)})`, 'g'),
        '<mark>$1</mark>'
      );
      return `<div class="search-result-item"
                   role="option">${highlighted}</div>`;
    }).join('');
  }

  results.classList.add('open');
}

// --- 正規表現の特殊文字をエスケープする関数 ---
function escapeRegex(str) {
  return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

// --- クリアボタン ---
clearBtn.addEventListener('click', () => {
  input.value = '';
  results.classList.remove('open');
  clearBtn.classList.remove('visible');
  input.focus();
});

// --- 外側クリックで閉じる ---
document.addEventListener('click', (e) => {
  if (!e.target.closest('.search-wrapper')) {
    results.classList.remove('open');
  }
});
