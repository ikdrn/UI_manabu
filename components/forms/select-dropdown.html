設計の意図

appearance: noneでブラウザデフォルトを消す
ネイティブの<select>はブラウザごとに見た目が異なります。appearance: noneでデフォルトスタイルをリセットし、SVGの背景画像で統一された矢印アイコンを表示します。

ネイティブ vs カスタムの使い分け
ネイティブ<select>はアクセシビリティとモバイル対応が組み込まれている利点があります。カスタムドロップダウンはデザインの自由度が高い反面、キーボードナビゲーションとARIAロールを自分で実装する必要があります。

キーボードナビゲーションは必須
カスタムドロップダウンでは↑↓で移動、Enterで選択、Escapeで閉じるキーボード操作を実装します。role="listbox"とrole="option"でスクリーンリーダーにもリストであることを伝えます。

フォーカス管理の重要性
ドロップダウンを開いた時にフォーカスを最初の選択肢に移動し、閉じた時にトリガーボタンに戻す。この「フォーカストラップ」がないと、キーボードユーザーは迷子になります。

/* --- ネイティブ <select> のスタイリング --- */
.native-select {
  /* appearance: none → ブラウザデフォルトの外観を完全にリセットする。
     これがないとOSごとに異なる矢印アイコンが表示される */
  appearance: none;
  -webkit-appearance: none; /* Safari対応 */

  /* padding-right → 矢印アイコン分のスペースを確保 */
  padding: 10px 36px 10px 12px;

  /* background-image → SVGをインラインで埋め込んで矢印アイコンにする。
     URLエンコードしたSVGをdata:URIとして指定する */
  background-image: url("data:image/svg+xml,...");
  background-repeat: no-repeat;
  background-position: right 10px center;
  background-size: 16px;
}

/* --- カスタムドロップダウンのオプションリスト --- */
.custom-select-options {
  /* position: absolute → ボタンの真下に浮かせて表示 */
  position: absolute;
  top: calc(100% + 4px);
  left: 0;
  right: 0;
  max-height: 220px;
  overflow-y: auto;
  /* z-index → 他の要素より手前に表示する */
  z-index: var(--z-dropdown);
  display: none; /* 初期状態は非表示 */
}

.custom-select-options.open {
  display: block; /* .openクラスで表示する */
}

/* .focused → キーボードナビゲーション中の現在位置を示す */
.custom-select-option.focused {
  background-color: var(--color-brand-50);
}

/* .selected → 選択済みの項目。チェックマークを表示 */
.custom-select-option.selected::after {
  content: '✓';
  float: right;
  color: var(--color-brand-500);
}

/* --- 矢印の回転アニメーション --- */
.custom-select-arrow {
  transition: transform 150ms;
}

/* aria-expanded="true"時に矢印を180度回転 */
.custom-select-trigger[aria-expanded="true"] .custom-select-arrow {
  transform: rotate(180deg);
}
// --- カスタムドロップダウンの実装 ---
const select = document.getElementById('custom-select-demo');
const trigger = select.querySelector('.custom-select-trigger');
const options = select.querySelector('.custom-select-options');
const optionEls = [...options.querySelectorAll('.custom-select-option')];
const textEl = trigger.querySelector('.custom-select-text');

let focusedIndex = -1; // 現在フォーカスされている項目のインデックス

// --- 開閉トグル ---
trigger.addEventListener('click', () => {
  const isOpen = options.classList.contains('open');
  if (isOpen) {
    closeDropdown();
  } else {
    openDropdown();
  }
});

function openDropdown() {
  options.classList.add('open');
  trigger.setAttribute('aria-expanded', 'true');
  // 選択済みの項目があればそこにフォーカス
  const selectedIdx = optionEls.findIndex(
    el => el.classList.contains('selected')
  );
  setFocus(selectedIdx >= 0 ? selectedIdx : 0);
}

function closeDropdown() {
  options.classList.remove('open');
  trigger.setAttribute('aria-expanded', 'false');
  clearFocus();
  trigger.focus(); // トリガーにフォーカスを戻す
}

// --- キーボードナビゲーション ---
trigger.addEventListener('keydown', (e) => {
  switch (e.key) {
    case 'ArrowDown':
    case 'ArrowUp':
      e.preventDefault();
      if (!options.classList.contains('open')) {
        openDropdown();
      }
      break;
    case 'Escape':
      closeDropdown();
      break;
  }
});

options.addEventListener('keydown', (e) => {
  switch (e.key) {
    case 'ArrowDown':
      e.preventDefault();
      // 次の項目に移動（最後なら最初に戻る）
      setFocus((focusedIndex + 1) % optionEls.length);
      break;
    case 'ArrowUp':
      e.preventDefault();
      // 前の項目に移動（最初なら最後に飛ぶ）
      setFocus(
        (focusedIndex - 1 + optionEls.length) % optionEls.length
      );
      break;
    case 'Enter':
    case ' ':
      e.preventDefault();
      selectOption(focusedIndex);
      closeDropdown();
      break;
    case 'Escape':
      closeDropdown();
      break;
  }
});

// --- フォーカス管理 ---
function setFocus(index) {
  clearFocus();
  focusedIndex = index;
  const el = optionEls[index];
  el.classList.add('focused');
  el.focus();
  // スクロール位置を調整
  el.scrollIntoView({ block: 'nearest' });
}

function clearFocus() {
  optionEls.forEach(el => el.classList.remove('focused'));
  focusedIndex = -1;
}

// --- 選択処理 ---
function selectOption(index) {
  optionEls.forEach(el => {
    el.classList.remove('selected');
    el.setAttribute('aria-selected', 'false');
  });
  const el = optionEls[index];
  el.classList.add('selected');
  el.setAttribute('aria-selected', 'true');
  textEl.textContent = el.textContent;
  textEl.classList.remove('custom-select-placeholder');
}

// クリックで選択
optionEls.forEach((el, i) => {
  el.addEventListener('click', () => {
    selectOption(i);
    closeDropdown();
  });
});

// 外側クリックで閉じる
document.addEventListener('click', (e) => {
  if (!e.target.closest('#custom-select-demo')) {
    closeDropdown();
  }
});
