ドラッグ&ドロップ

ソート可能リスト

ドラッグ操作でアイテムの順序を入れ替えられるリスト。ドラッグハンドル、視覚的フィードバック、ドロップインジケーターを備え、直感的な並べ替え操作を実現します。

  • 2 ワイヤーフレームを設計する
  • 3 デザインモックアップを作成する

設計の意図

なぜドラッグハンドルが必要なのか？
アイテム全体をドラッグ可能にすると、テキスト選択やリンククリックと干渉します。ハンドルを分離することで「ここを掴めば動かせる」と明示し、他の操作との衝突を防ぎます。

ドラッグ中の半透明表現
ドラッグ中のアイテムを半透明にすることで「これが移動中のアイテムだ」と視覚的に伝えます。同時にドロップ先にインジケーターを表示し、どこに挿入されるかを予測可能にします。

insertBefore()による自然な並べ替え
DOM操作のinsertBefore()を使うことで、ドラッグ＆ドロップの結果がリアルタイムでDOMに反映されます。仮想的な状態管理ではなく、実際のDOM構造が変わるため、直感的に理解しやすい実装です。

コード解説

<!-- ul要素 → 順序付きリストのコンテナ -->
<ul class="sortable-list" id="sortableList">

  <!-- draggable="true" → この要素をドラッグ可能にするHTML属性 -->
  <li class="sortable-item" draggable="true">

    <!-- ドラッグハンドル → 掴める場所を示すUI部品 -->
    <div class="drag-handle" aria-label="ドラッグして並べ替え">
      <div class="drag-handle-dots">
        <span></span><span></span>
        <span></span><span></span>
        <span></span><span></span>
      </div>
    </div>

    <!-- 番号バッジ → 現在の順番を表示 -->
    <span class="sortable-item-number">1</span>

    <!-- タスク内容 -->
    <span class="sortable-item-content">要件定義書を作成する</span>
  </li>
</ul>
/* ソート可能アイテムの基本スタイル */
.sortable-item {
  /* display: flex → 横並びレイアウト。ハンドル・番号・テキストを一行に */
  display: flex;
  align-items: center;
  gap: var(--space-3);
  padding: var(--space-3) var(--space-4);
  background: var(--color-bg-primary);
  border: 1px solid var(--color-border-secondary);
  border-radius: var(--radius-md);
  margin-bottom: var(--space-2);

  /* transition → ドラッグ時の変化をスムーズにする */
  transition: box-shadow 150ms, transform 150ms, opacity 150ms;

  /* user-select: none → テキスト選択を防止してドラッグ操作に専念させる */
  user-select: none;
}

/* ドラッグハンドル — cursor: grab で「掴める」ことを示す */
.drag-handle {
  cursor: grab;
  color: var(--color-text-tertiary);
}

.drag-handle:active {
  cursor: grabbing; /* 掴んでいる最中は拳マークに変化 */
}

/* ドラッグ中のアイテム — 半透明にして「移動中」を表現 */
.sortable-item.dragging {
  opacity: 0.4;
  transform: scale(0.98);
}

/* ドロップ位置のインジケーター — 上辺または下辺にブランドカラーの線を表示 */
.sortable-item.drag-over-top {
  border-top: 3px solid var(--color-brand-500);
}

.sortable-item.drag-over-bottom {
  border-bottom: 3px solid var(--color-brand-500);
}
// ソート可能リストの初期化
const list = document.getElementById('sortableList');
let draggedItem = null; // 現在ドラッグ中のアイテムを保持する変数

// dragstart → ドラッグが開始された時に発火するイベント
list.addEventListener('dragstart', (e) => {
  // closest() → イベント発生要素から最も近い親要素を探すメソッド
  draggedItem = e.target.closest('.sortable-item');
  draggedItem.classList.add('dragging');
  list.classList.add('has-dragging');

  // dataTransfer.effectAllowed → ドラッグ操作の種類を指定
  // 'move' はアイテムの移動を意味する
  e.dataTransfer.effectAllowed = 'move';
});

// dragover → ドラッグ中の要素がこの上を通過中に連続発火するイベント
list.addEventListener('dragover', (e) => {
  e.preventDefault(); // デフォルト動作を防止してdropを発火可能に
  const item = e.target.closest('.sortable-item');
  if (!item || item === draggedItem) return;

  // マウスのY座標とアイテム中央を比較して挿入位置を決定
  const rect = item.getBoundingClientRect();
  const midY = rect.top + rect.height / 2;

  clearIndicators();
  if (e.clientY < midY) {
    item.classList.add('drag-over-top');   // 上半分 → 上に挿入
  } else {
    item.classList.add('drag-over-bottom'); // 下半分 → 下に挿入
  }
});

// drop → ドラッグ要素がドロップされた時に発火するイベント
list.addEventListener('drop', (e) => {
  e.preventDefault();
  const item = e.target.closest('.sortable-item');
  if (!item || item === draggedItem) return;

  const rect = item.getBoundingClientRect();
  const midY = rect.top + rect.height / 2;

  // insertBefore() → 指定要素の前にノードを挿入するDOM API
  if (e.clientY < midY) {
    list.insertBefore(draggedItem, item);
  } else {
    list.insertBefore(draggedItem, item.nextSibling);
  }
  updateNumbers(); // 番号バッジを振り直す
});

// dragend → ドラッグ操作が終了した時に発火（成功/キャンセル問わず）
list.addEventListener('dragend', () => {
  draggedItem?.classList.remove('dragging');
  list.classList.remove('has-dragging');
  clearIndicators();
  draggedItem = null;
});

// ヘルパー: すべてのインジケーターを消去
function clearIndicators() {
  list.querySelectorAll('.sortable-item').forEach(item => {
    item.classList.remove('drag-over-top', 'drag-over-bottom');
  });
}

// ヘルパー: 番号バッジを現在の並び順で更新
function updateNumbers() {
  list.querySelectorAll('.sortable-item-number').forEach((num, i) => {
    num.textContent = i + 1;
  });
}
