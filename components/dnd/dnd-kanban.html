カンバンボード

タスクをカード形式で管理し、ドラッグ操作でカラム間を移動できるカンバンボード。3つのステータス列（未着手・進行中・完了）を持ち、直感的なタスク管理を実現します。

設計の意図

複数のドロップゾーンを持つ設計
各カラムが独立したドロップゾーンとして機能します。dragoverイベントでカラム全体をハイライトすることで、「ここにドロップできる」ことを明確に示します。

dataTransferによるカード情報の受け渡し
dragstart時にdataTransfer.setData()でカードのIDを保存し、drop時にgetData()で取得します。これにより、どのカードがどこに移動したかを追跡できます。

ステータスごとの色分け
未着手はグレー、進行中は青、完了は緑のドットで視覚的にステータスを区別します。色だけでなくテキストも併用してアクセシビリティに配慮します。

コード解説

<!-- カンバンボード → 3列のグリッドレイアウト -->
<div class="kanban-board" id="kanbanBoard">

  <!-- 各カラム → data-statusでステータスを識別 -->
  <div class="kanban-column" data-status="todo">
    <div class="kanban-column-header">
      <div class="kanban-column-title">
        <span class="status-dot status-dot--todo"></span>
        未着手
      </div>
      <span class="kanban-count">3</span>
    </div>

    <!-- カードリスト → ドロップゾーンとなる領域 -->
    <div class="kanban-cards">
      <!-- draggable="true" → ドラッグ可能に設定 -->
      <div class="kanban-card" draggable="true" data-id="1">
        <div class="kanban-card-title">タスク名</div>
        <div class="kanban-card-meta">
          <span class="kanban-priority kanban-priority--high">高</span>
          <span>期限日</span>
        </div>
      </div>
    </div>
  </div>

  <div class="kanban-column" data-status="progress">...</div>
  <div class="kanban-column" data-status="done">...</div>
</div>
/* カンバンボード → CSS Gridで3列均等配置 */
.kanban-board {
  display: grid;
  /* repeat(3, 1fr) → 3列を均等幅で配置 */
  grid-template-columns: repeat(3, 1fr);
  gap: var(--space-4);
}

/* カラム → 背景色で領域を区別 */
.kanban-column {
  background: var(--color-bg-secondary);
  border-radius: var(--radius-lg);
  padding: var(--space-4);
  min-height: 300px;
  /* transition → ドラッグオーバー時のハイライトを滑らかに */
  transition: background-color 150ms, box-shadow 150ms;
}

/* カード → 白背景で浮き上がらせる */
.kanban-card {
  background: var(--color-bg-primary);
  border: 1px solid var(--color-border-secondary);
  border-radius: var(--radius-md);
  padding: var(--space-3);
  cursor: grab;
  transition: box-shadow 150ms, transform 150ms, opacity 150ms;
}

/* ドラッグ中のカード → 半透明で移動中を表現 */
.kanban-card.dragging {
  opacity: 0.4;
  transform: scale(0.95);
}

/* ドラッグオーバー時のカラムハイライト
   → inset box-shadowで内側に枠線を表示 */
.kanban-column.drag-over {
  background-color: var(--color-brand-50);
  box-shadow: inset 0 0 0 2px var(--color-brand-300);
}
const board = document.getElementById('kanbanBoard');
let draggedCard = null;

// dragstart → カードのドラッグ開始
board.addEventListener('dragstart', (e) => {
  draggedCard = e.target.closest('.kanban-card');
  if (!draggedCard) return;

  // dataTransfer.setData() → ドラッグ中にデータを保持する仕組み
  // カードのdata-id属性値を文字列として保存
  e.dataTransfer.setData('text/plain', draggedCard.dataset.id);
  e.dataTransfer.effectAllowed = 'move';

  // requestAnimationFrame → 次のフレームで実行
  // ドラッグ画像が生成された後にクラスを追加するため
  requestAnimationFrame(() => {
    draggedCard.classList.add('dragging');
  });
});

// dragover → カラム上をドラッグ中に連続発火
board.addEventListener('dragover', (e) => {
  e.preventDefault(); // dropイベントを発火させるために必須
  const column = e.target.closest('.kanban-column');
  if (!column) return;

  // 他のカラムのハイライトを消してから、現在のカラムをハイライト
  clearColumnHighlights();
  column.classList.add('drag-over');
});

// dragleave → ドラッグ要素がカラムから出た時
board.addEventListener('dragleave', (e) => {
  const column = e.target.closest('.kanban-column');
  if (column) column.classList.remove('drag-over');
});

// drop → カラムにカードがドロップされた時
board.addEventListener('drop', (e) => {
  e.preventDefault();
  const column = e.target.closest('.kanban-column');
  if (!column || !draggedCard) return;

  // カードリスト要素を取得してカードを追加
  const cardList = column.querySelector('.kanban-cards');
  cardList.appendChild(draggedCard);

  clearColumnHighlights();
  updateCounts(); // 各カラムのカード数を更新
});

// dragend → ドラッグ終了時のクリーンアップ
board.addEventListener('dragend', () => {
  if (draggedCard) draggedCard.classList.remove('dragging');
  clearColumnHighlights();
  draggedCard = null;
});

// ヘルパー: カラムのハイライトをすべて消去
function clearColumnHighlights() {
  board.querySelectorAll('.kanban-column').forEach(col => {
    col.classList.remove('drag-over');
  });
}

// ヘルパー: 各カラムのカード数バッジを更新
function updateCounts() {
  board.querySelectorAll('.kanban-column').forEach(col => {
    const count = col.querySelectorAll('.kanban-card').length;
    col.querySelector('.kanban-count').textContent = count;
  });
}
