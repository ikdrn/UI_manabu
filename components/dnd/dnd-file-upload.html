ドラッグ&ドロップ

ファイルドロップアップロード

ファイルをドラッグ&ドロップでアップロードできるゾーン。破線のボーダーでドロップ領域を示し、ドロップ後にファイル一覧と画像プレビューを表示します。

プレビュー

ファイルをここにドラッグ、またはクリックして選択

PNG, JPG, PDF, 最大10MBまで

    設計の意図

    破線ボーダーが「ここにドロップできる」を伝える
    破線（dashed）のボーダーは「まだ何かが入る余地がある」という視覚的メタファーです。実線と異なり「完結していない」印象を与え、ユーザーにアクションを促します。

    dragenter/dragleave/dropの3段階イベント
    dragenterで「入った」、dragleaveで「出た」、dropで「離した」を検知します。ブラウザのデフォルト動作（ファイルを開く）をpreventDefault()で防ぎ、アプリ側で処理します。

    FileReaderによる画像プレビュー
    FileReader APIのreadAsDataURL()で画像をBase64文字列に変換し、imgタグのsrcに直接設定します。サーバーにアップロードする前にプレビューを表示でき、ユーザーの確認操作を支援します。

    const dropZone = document.getElementById('dropZone');
const fileInput = document.getElementById('fileInput');
const fileList = document.getElementById('fileList');

// クリックでファイル選択ダイアログを開く
dropZone.addEventListener('click', () => fileInput.click());

// dragenter → ファイルがゾーンに入った時
dropZone.addEventListener('dragenter', (e) => {
  e.preventDefault();
  dropZone.classList.add('drag-active');
});

// dragover → ファイルがゾーン上にある間、連続発火
dropZone.addEventListener('dragover', (e) => {
  e.preventDefault(); // ブラウザがファイルを開くのを防ぐ
});

// dragleave → ファイルがゾーンから出た時
dropZone.addEventListener('dragleave', () => {
  dropZone.classList.remove('drag-active');
});

// drop → ファイルがドロップされた時
dropZone.addEventListener('drop', (e) => {
  e.preventDefault();
  dropZone.classList.remove('drag-active');

  // e.dataTransfer.files → ドロップされたファイルのリスト
  const files = e.dataTransfer.files;
  handleFiles(files);
});

// input変更時 → クリックでファイル選択した場合
fileInput.addEventListener('change', (e) => {
  handleFiles(e.target.files);
});

// ファイル処理の共通関数
function handleFiles(files) {
  Array.from(files).forEach(file => {
    const li = document.createElement('li');
    li.className = 'file-item';

    // 画像プレビューの生成
    const preview = document.createElement('div');
    preview.className = 'file-preview';

    // file.type.startsWith('image/') → 画像ファイルかどうかを判定
    if (file.type.startsWith('image/')) {
      // FileReader → ファイルの内容を非同期で読み取るAPI
      const reader = new FileReader();
      reader.onload = (e) => {
        const img = document.createElement('img');
        // e.target.result → Base64エンコードされた画像データ
        img.src = e.target.result;
        preview.appendChild(img);
      };
      // readAsDataURL → ファイルをBase64文字列として読み取る
      reader.readAsDataURL(file);
    } else {
      preview.textContent = getFileExtension(file.name);
    }

    // ファイル情報
    const info = document.createElement('div');
    info.className = 'file-info';
    info.innerHTML = `
      <div class="file-name">${file.name}</div>
      <div class="file-size">${formatFileSize(file.size)}</div>
    `;

    // 削除ボタン
    const removeBtn = document.createElement('button');
    removeBtn.className = 'file-remove';
    removeBtn.innerHTML = '&times;';
    removeBtn.onclick = () => li.remove();

    li.append(preview, info, removeBtn);
    fileList.appendChild(li);
  });
}

// ファイルサイズを人間が読みやすい形式に変換
function formatFileSize(bytes) {
  if (bytes === 0) return '0 B';
  const units = ['B', 'KB', 'MB', 'GB'];
  // Math.log(bytes) / Math.log(1024) → 1024の何乗かを計算
  const i = Math.floor(Math.log(bytes) / Math.log(1024));
  return (bytes / Math.pow(1024, i)).toFixed(1) + ' ' + units[i];
}

// ファイル拡張子を取得
function getFileExtension(name) {
  return name.split('.').pop().toUpperCase();
}
